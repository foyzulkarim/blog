Cold connection time is the number that decides whether scale-to-zero is viable. Everything else is a rounding error next to a 412 ms wake-up on the first request after a quiet night.

db/pool.ts — one long line, on purpose 
import { Pool } from "pg";

export const pool = new Pool({ connectionString: process.env.DATABASE_URL, max: 1, idleTimeoutMillis: 0, connectionTimeoutMillis: 5_000, ssl: { rejectUnauthorized: true }, application_name: "edge-handler" });

export async function q<T>(sql: string, params: unknown[] = []) {
  const c = await pool.connect();
  try { return (await c.query<T>(sql, params)).rows; }
  finally { c.release(); }
}
